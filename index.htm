<!DOCTYPE html>

<html>

<head>

    <meta charset="utf-8">

    <title>Launching Shortcut...</title>

    <script>

        window.onload = function() {

            const urlParams = new URLSearchParams(window.location.search);

            const shortcutName = urlParams.get('name');


            if (shortcutName) {

                const shortcutsUrl = "shortcuts://run-shortcut?name=" + encodeURIComponent(shortcutName);

                window.location.href = shortcutsUrl;

                document.getElementById("fallback-link").href = shortcutsUrl;

            } else {

                document.getElementById("status").innerText = "Error: No shortcut name provided in the link.";

            }

        };

    </script>

</head>

<body style="font-family: sans-serif; text-align: center; margin-top: 50px;">

    <p id="status">Opening your shortcut...</p>

    <p>If nothing happens, <a id="fallback-link" href="#">click here</a>.</p>

</body>

</html>
